>001971:     /// RAII guard for one unique `.slhw` model file.
>001972:     ///
>001973:     /// Every instance creates its own `tempfile::TempDir` and a unique file
>001974:     /// inside it, so parallel tests cannot collide on the same path.  Drop
>001975:     /// releases the C handle *before* the temp directory is removed, so the
>001976:     /// file is never deleted while the loaded model is still live.

>001978:         handle: *mut SlhaModel,
>001979:         model: LearnedModel,

>001988:                 .prefix(&format!("slha_c_{counter}_"))
>001989:                 .tempdir()

>002024:     impl Drop for TempModel {
>002025:         fn drop(&mut self) {
>002026:             if !self.handle.is_null() {

>002028:                 // released exactly once here.
>002029:                 unsafe {
>002030:                     let _ = slha_weights_release(self.handle);
>002031:                 }
>002032:                 self.handle = std::ptr::null_mut();
>002033:             }
